import { z } from 'zod'
import * as React from 'react'
import type { BlockDefinition } from '@pwfabric/core'
import type { BlockComponent } from '../../component-types.js'

// ============================================================
// SCHEMA
// ============================================================

export const progressPropsSchema = z.object({
  value: z.number().min(0).max(100).default(50),
  max: z.number().min(1).default(100),
  showLabel: z.boolean().default(false),
  labelFormat: z.enum(['percent', 'value', 'fraction']).default('percent'),
  size: z.enum(['sm', 'md', 'lg']).default('md'),
  color: z.string().optional(),
  animated: z.boolean().default(false),
})

export type ProgressProps = z.infer<typeof progressPropsSchema>

// ============================================================
// DEFINITION
// ============================================================

export const progressDefinition: BlockDefinition = {
  type: 'progress',
  name: 'Progress',
  category: 'data',
  icon: '▬',
  schema: progressPropsSchema,
  defaultProps: {
    value: 50,
    max: 100,
    showLabel: false,
    labelFormat: 'percent',
    size: 'md',
    animated: false,
  },
  defaultLayout: { span: { xs: 4, sm: 8, md: 8, lg: 12 } },
  canHaveChildren: false,
}

// ============================================================
// COMPONENT
// ============================================================

export const ProgressBlock: BlockComponent = ({ props, block }) => {
  const max = (props.max as number) ?? 100
  const value = Math.min(Math.max((props.value as number) ?? 0, 0), max)
  const pct = max > 0 ? (value / max) * 100 : 0

  const sizeMap: Record<string, string> = { sm: '6px', md: '10px', lg: '16px' }
  const colorMap: Record<string, string> = {
    primary: 'var(--pw-color-primary)',
    accent: 'var(--pw-color-accent)',
    success: 'var(--pw-color-success)',
    warning: 'var(--pw-color-warning)',
    error: 'var(--pw-color-error)',
  }
  const trackStyle: React.CSSProperties = {
    width: '100%',
    borderRadius: '999px',
    overflow: 'hidden',
    backgroundColor: 'var(--pw-color-border)',
    height: sizeMap[(props.size as string) ?? 'md'] ?? '10px',
  }
  const fillStyle: React.CSSProperties = {
    height: '100%',
    borderRadius: '999px',
    width: `${pct}%`,
    transition: props.animated ? 'width var(--pw-duration-slow) var(--pw-easing)' : undefined,
    backgroundColor: colorMap[(props.color as string) ?? 'primary'] ?? colorMap['primary'],
  }
  const label =
    props.labelFormat === 'value'
      ? `${value}/${max}`
      : props.labelFormat === 'none'
        ? undefined
        : `${Math.round(pct)}%`

  return (
    <div
      id={block.id}
      data-block-id={block.id}
      style={{ display: 'flex', flexDirection: 'column', gap: '4px', ...block.style }}
      data-block-type="progress"
    >
      {!!props.showLabel && label && (
        <div
          style={{
            fontSize: 'var(--pw-font-size-xs)',
            fontWeight: 500,
            color: 'var(--pw-color-text)',
            textAlign: 'right',
          }}
        >
          {label}
        </div>
      )}
      <div
        role="progressbar"
        aria-valuenow={value}
        aria-valuemin={0}
        aria-valuemax={max}
        style={trackStyle}
      >
        <div style={fillStyle} />
      </div>
    </div>
  )
}
